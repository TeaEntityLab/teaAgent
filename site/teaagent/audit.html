131    def _apply_audit_level(self, payload: dict[str, Any]) -> dict[str, Any]:
132        """Apply tiered audit level filtering to payload.
133
134        L0: Metrics only (minimal metadata)
135        L1: Metadata (event types, timestamps, no arguments)
136        L2: Redacted payload (default, sensitive data redacted)
137        L3: Full local trace (all data, encrypted at rest)
138        """
139        if self._audit_level == 'L0':
140            # Only keep basic metrics
141            return {
142                'event_type': payload.get('event_type'),
143                'timestamp': payload.get('timestamp'),
144            }
145        elif self._audit_level == 'L1':
146            # Keep metadata but remove detailed arguments/results
147            filtered = dict(payload)
148            # Remove sensitive detailed fields
149            for key in ['arguments', 'result', 'content', 'output', 'input']:
150                filtered.pop(key, None)
151            return filtered
152        elif self._audit_level == 'L2':
153            # Default redacted payload (already handled by redact_audit_payload)
154            return payload
155        elif self._audit_level == 'L3':
156            # Full trace - no additional filtering
157            return payload
158        else:
159            # Default to L2 behavior
160            return payload

203    def record(self, event_type: str, run_id: str, **payload: Any) -> AuditEvent:
204        # Apply tiered audit level filtering
205        filtered_payload = self._apply_audit_level(payload)
206
207        event = AuditEvent(
208            event_type=event_type,
209            run_id=run_id,
210            payload=redact_audit_payload(
211                filtered_payload, string_patterns=self._string_patterns
212            ),
213        )
214        with self._lock:
215            self.events.append(event)
216            if self.path is not None and self._disk_error is None:
217                prev = self._prev_hash
218                canonical = json.dumps(
219                    {
220                        'event_id': event.event_id,
221                        'event_type': event.event_type,
222                        'run_id': event.run_id,
223                        'created_at': event.created_at,
224                        'payload': event.payload,
225                        'prev_hash': prev,
226                    },
227                    sort_keys=True,
228                    separators=(',', ':'),
229                )
230                current_hash = hashlib.sha256(canonical.encode('utf-8')).hexdigest()
231                try:
232                    append_jsonl_line(
233                        self.path,
234                        event.to_json(prev_hash=prev, event_hash=current_hash),
235                    )
236                    self._prev_hash = current_hash
237                    secure_audit_file(self.path)
238                except OSError as exc:
239                    self._disk_error = exc
240                    err_event = AuditEvent(
241                        event_type='_disk_write_error',
242                        run_id=event.run_id,
243                        payload={'error': str(exc), 'errno': exc.errno},
244                    )
245                    self.events.append(err_event)
246            sinks = list(self._sinks)
247        for sink in sinks:
248            with contextlib.suppress(Exception):
249                sink(event)
250        return event

252    def verify_chain_integrity(self) -> dict[str, Any]:
253        """Verify the hash chain integrity of the audit log.
254
255        Returns:
256            Dict with 'valid' (bool), 'total_events' (int), and 'errors' (list[str])
257        """
258        if not self.path or not self.path.is_file():
259            return {
260                'valid': False,
261                'total_events': 0,
262                'errors': ['No audit file found'],
263            }
264
265        try:
266            lines = self.path.read_text(encoding='utf-8').splitlines()
267            events = []
268            for line in lines:
269                if line.strip():
270                    events.append(json.loads(line))
271
272            if not events:
273                return {'valid': True, 'total_events': 0, 'errors': []}
274
275            errors = []
276            prev_hash = _GENESIS_HASH
277            for i, event in enumerate(events):
278                event_hash = event.get('hash')
279                event_prev_hash = event.get('prev_hash')
280
281                if event_prev_hash != prev_hash:
282                    errors.append(
283                        f'Event {i}: prev_hash mismatch (expected {prev_hash}, got {event_prev_hash})'
284                    )
285
286                # Verify the hash matches the content
287                canonical = json.dumps(
288                    {
289                        'event_id': event.get('event_id'),
290                        'event_type': event.get('event_type'),
291                        'run_id': event.get('run_id'),
292                        'created_at': event.get('created_at'),
293                        'payload': event.get('payload'),
294                        'prev_hash': event_prev_hash,
295                    },
296                    sort_keys=True,
297                    separators=(',', ':'),
298                )
299                computed_hash = hashlib.sha256(canonical.encode('utf-8')).hexdigest()
300
301                if event_hash != computed_hash:
302                    errors.append(
303                        f'Event {i}: hash mismatch (expected {computed_hash}, got {event_hash})'
304                    )
305
306                prev_hash = event_hash or computed_hash
307
308            return {
309                'valid': len(errors) == 0,
310                'total_events': len(events),
311                'errors': errors,
312            }
313        except Exception as exc:
314            return {
315                'valid': False,
316                'total_events': 0,
317                'errors': [f'Verification failed: {exc}'],
318            }

321def redact_audit_payload(
322    payload: dict[str, Any],
323    *,
324    string_patterns: Any = None,
325) -> dict[str, Any]:
326    patterns = (
327        string_patterns if string_patterns is not None else SENSITIVE_STRING_PATTERNS
328    )
329    redacted: dict[str, Any] = {}
330    for key, value in payload.items():
331        if key == 'arguments' and isinstance(value, dict):
332            redacted[key] = redact_tool_arguments(value, string_patterns=patterns)
333        elif key == 'result' and isinstance(value, dict):
334            redacted[key] = redact_tool_result(value, string_patterns=patterns)
335        else:
336            redacted[key] = redact_audit_value(key, value, string_patterns=patterns)
337    return redacted

359def redact_tool_argument_value(
360    key: str, value: Any, *, string_patterns: Any = None
361) -> Any:
362    if is_sensitive_key(key) or key in SENSITIVE_ARGUMENT_KEYS:
363        return AUDIT_REDACTED
364    if isinstance(value, dict):
365        return {
366            str(child_key): redact_tool_argument_value(
367                str(child_key), child_value, string_patterns=string_patterns
368            )
369            for child_key, child_value in value.items()
370        }
371    if isinstance(value, list):
372        return [
373            redact_tool_argument_value('', item, string_patterns=string_patterns)
374            for item in value
375        ]
376    if isinstance(value, tuple):
377        return [
378            redact_tool_argument_value('', item, string_patterns=string_patterns)
379            for item in value
380        ]
381    return redact_audit_value(key, value, string_patterns=string_patterns)

420def redact_audit_value(key: str, value: Any, *, string_patterns: Any = None) -> Any:
421    # Only redact string / bytes values by key sensitivity.
422    # Numeric, bool, and None values are telemetry data and are never sensitive.
423    if is_sensitive_key(key) and isinstance(value, (str, bytes)):
424        return AUDIT_REDACTED
425    if isinstance(value, dict):
426        return {
427            str(child_key): redact_audit_value(
428                str(child_key), child_value, string_patterns=string_patterns
429            )
430            for child_key, child_value in value.items()
431        }
432    if isinstance(value, list):
433        return [
434            redact_audit_value('', item, string_patterns=string_patterns)
435            for item in value
436        ]
437    if isinstance(value, tuple):
438        return [
439            redact_audit_value('', item, string_patterns=string_patterns)
440            for item in value
441        ]
442    if isinstance(value, str):
443        patterns = (
444            string_patterns
445            if string_patterns is not None
446            else SENSITIVE_STRING_PATTERNS
447        )
448        redacted = redact_sensitive_string(value, patterns=patterns)
449        if len(redacted) > MAX_AUDIT_STRING_LENGTH:
450            return redacted[:MAX_AUDIT_STRING_LENGTH] + AUDIT_TRUNCATED
451        return redacted
452    return value

455def redact_sensitive_string(value: str, *, patterns: Any = None) -> str:
456    active = patterns if patterns is not None else SENSITIVE_STRING_PATTERNS
457    redacted = value
458    for pattern, replacement in active:
459        redacted = pattern.sub(replacement, redacted)
460    return redacted
