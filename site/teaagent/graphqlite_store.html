 40class GraphQLiteGraphStore:
 41    """GraphQLite-backed store for Graph RAG entity and relation data."""
 42
 43    def __init__(
 44        self,
 45        config: Optional[GraphQLiteConfig] = None,
 46        *,
 47        graph_factory: Optional[GraphFactory] = None,
 48    ) -> None:
 49        self.config = config or GraphQLiteConfig()
 50        try:
 51            self.graph = (graph_factory or load_graphqlite_graph)(self.config.database)
 52        except (GraphQLiteUnavailableError, GraphQLiteRuntimeError) as exc:
 53            import sys
 54
 55            print(
 56                f'[TeaAgent WARNING] GraphQLite runtime is unavailable: {exc}. '
 57                f'Gracefully degrading to file-level semantic index & hybrid search fallback.',
 58                file=sys.stderr,
 59            )
 60            self.graph = DummyKnowledgeGraph()
 61
 62    def upsert_document(self, document: Document) -> None:
 63        self.graph.upsert_node(
 64            document.doc_id,
 65            {
 66                'doc_id': document.doc_id,
 67                'text': document.text,
 68                'source': document.source,
 69                **document.metadata,
 70            },
 71            label='Document',
 72        )
 73
 74    def upsert_edge(self, edge: GraphEdge) -> None:
 75        self.graph.upsert_node(edge.source, {'name': edge.source}, label='Entity')
 76        self.graph.upsert_node(edge.target, {'name': edge.target}, label='Entity')
 77        self.graph.upsert_edge(
 78            edge.source,
 79            edge.target,
 80            {'relation': edge.relation, 'document_ids': list(edge.document_ids)},
 81            rel_type=edge.relation.upper(),
 82        )
 83
 84    def query(self, cypher: str, params: Optional[dict[str, Any]] = None) -> Any:
 85        if params is not None:
 86            return self.graph.query(cypher, params=params)
 87        return self.graph.query(cypher)
 88
 89    def sync_from_knowledge_graph(self, graph: KnowledgeGraph) -> None:
 90        for document in graph.all_documents():
 91            self.upsert_document(document)
 92        for edge in graph.all_edges():
 93            self.upsert_edge(edge)

 96def load_graphqlite_graph(database: str) -> Any:
 97    ensure_sqlite_extension_loading()
 98    try:
 99        from graphqlite import Graph
100    except ImportError as exc:  # pragma: no cover - depends on optional runtime package
101        raise GraphQLiteUnavailableError(
102            'graphqlite is required for GraphQLiteGraphStore. Install requirements.txt or pyproject dependencies.'
103        ) from exc
104    try:
105        return Graph(database)
106    except RuntimeError as exc:
107        raise GraphQLiteRuntimeError(
108            'graphqlite is installed, but the current Python sqlite3 runtime cannot load SQLite extensions. '
109            'Use a Python build with sqlite3.enable_load_extension support, such as a Homebrew Python on macOS.'
110        ) from exc

113def ensure_sqlite_extension_loading() -> None:
114    """Use pysqlite3 when the platform sqlite3 lacks extension loading."""
115
116    import sqlite3
117
118    connection = sqlite3.connect(':memory:')
119    try:
120        if hasattr(connection, 'enable_load_extension'):
121            return
122    finally:
123        connection.close()
124
125    try:
126        import pysqlite3
127    except (
128        ImportError
129    ) as exc:  # pragma: no cover - dependency/runtime environment specific
130        raise GraphQLiteRuntimeError(
131            'graphqlite requires sqlite extension loading, but the current sqlite3 runtime does not support it. '
132            'Install pysqlite3 or use a Python build with sqlite3.enable_load_extension support.'
133        ) from exc
134
135    sys.modules['sqlite3'] = pysqlite3

138def check_graphqlite_runtime(database: str = ':memory:') -> tuple[bool, str]:
139    try:
140        graph = load_graphqlite_graph(database)
141        graph.upsert_node('teaagent_smoke', {'name': 'TeaAgent'}, label='SmokeTest')
142        graph.query('MATCH (n:SmokeTest) RETURN n.name')
143    except (GraphQLiteUnavailableError, GraphQLiteRuntimeError) as exc:
144        return False, str(exc)
145    return True, 'graphqlite runtime is available'
