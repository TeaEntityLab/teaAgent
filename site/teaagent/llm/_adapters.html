 39def _extract_claude_tool_calls(response: dict[str, Any]) -> list[LLMToolCall]:
 40    calls = []
 41    for block in response.get('content', []):
 42        if isinstance(block, dict) and block.get('type') == 'tool_use':
 43            calls.append(
 44                LLMToolCall(
 45                    tool_name=str(block.get('name', '')),
 46                    tool_input=dict(block.get('input') or {}),
 47                    call_id=str(block.get('id', '')),
 48                )
 49            )
 50    return calls

 53def _extract_openai_tool_calls(response: dict[str, Any]) -> list[LLMToolCall]:
 54    calls: list[LLMToolCall] = []
 55    choices = response.get('choices', [])
 56    if not choices:
 57        return calls
 58    message = choices[0].get('message', {})
 59    for tc in message.get('tool_calls') or []:
 60        fn = tc.get('function', {})
 61        try:
 62            args = json.loads(fn.get('arguments', '{}'))
 63        except Exception:
 64            args = {}
 65        calls.append(
 66            LLMToolCall(
 67                tool_name=str(fn.get('name', '')),
 68                tool_input=args,
 69                call_id=str(tc.get('id', '')),
 70            )
 71        )
 72    return calls

 75def _extract_gemini_tool_calls(response: dict[str, Any]) -> list[LLMToolCall]:

 77    for candidate in response.get('candidates', []):

 79            fc = part.get('functionCall')

 85                    )

103def _extract_gemini_safety(response: dict[str, Any]) -> 'LLMSafetyBlock | None':
104    for candidate in response.get('candidates', []):
105        if candidate.get('finishReason') == 'SAFETY':
106            for rating in candidate.get('safetyRatings', []):
107                if rating.get('blocked'):
108                    cat = _GEMINI_SAFETY_CATEGORY_MAP.get(
109                        rating.get('category', ''), SafetyCategory.OTHER
110                    )
111                    return LLMSafetyBlock(blocked=True, category=cat)
112            return LLMSafetyBlock(blocked=True)
113    return None

116class OpenAICompatibleAdapter:
117    def __init__(
118        self,
119        config: ProviderConfig,
120        *,
121        transport: Optional[HTTPTransport] = None,
122        timeout: int = 60,
123        retry_config: Optional[LLMRetryConfig] = None,
124        streaming_lines: Optional[list[bytes]] = None,
125    ) -> None:
126        self.config = config
127        self.provider = config.name
128        self.transport = transport or UrllibHTTPTransport()
129        self.timeout = timeout
130        self.retry_config = retry_config or DEFAULT_RETRY_CONFIG
131        self._streaming_lines = streaming_lines
132
133    def _prepare_payload(self, request: LLMRequest, model: str) -> dict[str, Any]:
134        messages = []
135        if request.system:
136            messages.append({'role': 'system', 'content': request.system})
137        messages.extend(
138            {'role': message.role, 'content': message.content}
139            for message in request.messages
140        )
141        payload: dict[str, Any] = {
142            'model': model,
143            'messages': messages,
144            'max_tokens': request.max_tokens,
145            'temperature': request.temperature,
146        }
147        if request.tools:
148            payload['tools'] = [
149                {
150                    'type': 'function',
151                    'function': {
152                        'name': t.name,
153                        'description': t.description,
154                        'parameters': t.input_schema,
155                    },
156                }
157                for t in request.tools
158            ]
159        if request.response_format and self._supports_response_format():
160            payload['response_format'] = request.response_format
161        if request.stream:
162            payload['stream'] = True
163        return payload
164
165    def complete(self, request: LLMRequest) -> LLMResponse:
166        model = request.model or self.config.resolved_model()
167        payload = self._prepare_payload(request, model)
168        if request.stream:
169            return self._complete_streaming(request, model, payload)
170        try:
171            response = self._post_chat_completions(payload)
172        except LLMHTTPError as exc:
173            if not (
174                request.response_format and _supports_response_format_fallback(exc)
175            ):
176                raise
177            payload = dict(payload)
178            payload.pop('response_format', None)
179            response = self._post_chat_completions(payload)
180        tool_calls = _extract_openai_tool_calls(response)
181        try:
182            content = _extract_openai_content(self.provider, response)
183        except LLMResponseFormatError:
184            if tool_calls:
185                content = _native_tool_calls_to_decision_text(tool_calls)
186            else:
187                raise
188        usage = response.get('usage', {})
189        return LLMResponse(
190            provider=self.provider,
191            model=model,
192            content=content,
193            raw=response,
194            input_tokens=usage.get('prompt_tokens', 0),
195            output_tokens=usage.get('completion_tokens', 0),
196            tool_calls=tool_calls,
197        )
198
199    def _supports_response_format(self) -> bool:
200        return self.provider not in {'opencodezen-go', 'opencodezen'}
201
202    def _post_chat_completions(self, payload: dict[str, Any]) -> dict[str, Any]:
203        return _call_with_retry(
204            self.provider,
205            lambda: self.transport.post_json(
206                f'{self.config.resolved_base_url()}/chat/completions',
207                self._headers(),
208                payload,
209                timeout=self.timeout,
210            ),
211            self.retry_config,
212        )
213
214    def _iter_streaming_lines(self, payload: dict[str, Any]) -> Iterator[bytes]:
215        if self._streaming_lines is not None:
216            for line in self._streaming_lines:
217                yield line if isinstance(line, bytes) else line.encode('utf-8')
218            return
219        body = json.dumps(payload).encode('utf-8')
220        url = f'{self.config.resolved_base_url()}/chat/completions'
221        headers = {
222            'content-type': 'application/json',
223            'user-agent': 'TeaAgent',
224            **self._headers(),
225        }
226        req = urllib_request.Request(url, data=body, headers=headers, method='POST')
227        try:
228            ssl_context = build_ssl_context_from_env()
229            request_kwargs: dict[str, Any] = {'timeout': self.timeout}
230            if ssl_context is not None:
231                request_kwargs['context'] = ssl_context
232            with urllib_request.urlopen(req, **request_kwargs) as resp:
233                yield from resp
234        except HTTPError as exc:
235            detail = exc.read().decode('utf-8', errors='replace')
236            raise LLMHTTPError(
237                f'HTTP {exc.code}: {detail}', status_code=exc.code
238            ) from exc
239        except URLError as exc:
240            raise LLMHTTPError(f'HTTP request failed: {exc.reason}') from exc
241
242    def _complete_streaming(
243        self, request: LLMRequest, model: str, payload: dict[str, Any]
244    ) -> LLMResponse:
245        chunks: list[str] = []
246        input_tokens = 0
247        output_tokens = 0
248
249        def _handle(parsed: dict[str, Any]) -> None:
250            nonlocal input_tokens, output_tokens
251            usage = parsed.get('usage')
252            if usage:
253                input_tokens = usage.get('prompt_tokens', 0)
254                output_tokens = usage.get('completion_tokens', 0)
255            delta = _first_choice_delta(self.provider, parsed)
256            chunk = delta.get('content', '')
257            if chunk and request.on_chunk:
258                request.on_chunk(chunk)
259            if chunk:
260                chunks.append(chunk)
261
262        consume_sse_json_chunks(
263            self._iter_streaming_lines(payload),
264            on_data=_handle,
265        )
266        return LLMResponse(
267            provider=self.provider,
268            model=model,
269            content=''.join(chunks),
270            raw={},
271            input_tokens=input_tokens,
272            output_tokens=output_tokens,
273        )
274
275    def _headers(self) -> dict[str, str]:
276        headers = {'authorization': f'Bearer {self.config.resolved_api_key()}'}
277        if self.provider == 'openrouter':
278            if os.environ.get('OPENROUTER_HTTP_REFERER'):
279                headers['HTTP-Referer'] = os.environ['OPENROUTER_HTTP_REFERER']
280            if os.environ.get('OPENROUTER_APP_TITLE'):
281                headers['X-Title'] = os.environ['OPENROUTER_APP_TITLE']
282        headers.update(_provider_extra_headers(self.provider))
283        return headers

286class WorkersAIAdapter(OpenAICompatibleAdapter):
287    """Adapter for Cloudflare Workers AI / AI Gateway.
288
289    Workers AI has several quirks vs standard OpenAI-compatible endpoints:
290    - AI Gateway's workers-ai provider route strips the ``@cf/`` model prefix.
291    - Structured output (``json_schema``) is unstable and often 500s.
292    - Streaming + response_format may not mix.
293    """
294
295    def __init__(
296        self,
297        config: ProviderConfig,
298        *,
299        transport: Optional[HTTPTransport] = None,
300        timeout: int = 60,
301        retry_config: Optional[LLMRetryConfig] = None,
302        streaming_lines: Optional[list[bytes]] = None,
303        disable_tools: bool = False,
304    ) -> None:
305        super().__init__(
306            config,
307            transport=transport,
308            timeout=timeout,
309            retry_config=retry_config,
310            streaming_lines=streaming_lines,
311        )
312        self.disable_tools = disable_tools
313
314    def _prepare_payload(self, request: LLMRequest, model: str) -> dict[str, Any]:
315        # AI Gateway's workers-ai provider endpoint often expects @cf/ to be
316        # stripped, otherwise it returns "Invalid provider" (HTTP 400).
317        # Direct Workers AI API DOES require @cf/.
318        base_url = self.config.resolved_base_url()
319        if 'gateway.ai.cloudflare.com' in base_url and model.startswith('@cf/'):
320            model = model[4:]
321
322        # Workers AI structured output (json_schema) is extremely unstable
323        # and often 500s. Default to removing it and relying on prompt
324        # instructions, unless forced via env var.
325        force_json_object = (
326            os.environ.get('TEAAGENT_WORKERS_AI_FORCE_JSON_OBJECT') == '1'
327        )
328
329        if request.response_format and not force_json_object:
330            request = LLMRequest(
331                messages=request.messages,
332                model=request.model,
333                system=(request.system or '')
334                + '\n\nIMPORTANT: You must respond with a valid JSON object matching the requested schema. No prose.',
335                max_tokens=request.max_tokens,
336                temperature=request.temperature,
337                stream=request.stream,
338                on_chunk=request.on_chunk,
339                tools=request.tools,
340                response_format=None,  # Strip it
341            )
342
343        payload = super()._prepare_payload(request, model)
344
345        if force_json_object:
346            payload['response_format'] = {'type': 'json_object'}
347
348        # Ensure stream and response_format never mix in Workers AI
349        if payload.get('stream') and 'response_format' in payload:
350            del payload['response_format']
351
352        # If tools cause 500s, allow disabling them entirely.
353        if self.disable_tools and 'tools' in payload:
354            del payload['tools']
355
356        return payload

359def _provider_extra_headers(provider: str) -> dict[str, str]:
360    env_prefix = provider.upper().replace('-', '_')
361    extra = os.environ.get(f'{env_prefix}_EXTRA_HEADERS')
362    if not extra:
363        return {}
364    try:
365        parsed = json.loads(extra)
366    except json.JSONDecodeError as exc:
367        raise LLMHTTPError(
368            f'invalid {env_prefix}_EXTRA_HEADERS JSON: {exc.msg}'
369        ) from exc
370    if not isinstance(parsed, dict):
371        raise LLMHTTPError(f'{env_prefix}_EXTRA_HEADERS must be a JSON object')
372    normalized: dict[str, str] = {}
373    for key, value in parsed.items():
374        if not isinstance(key, str):
375            raise LLMHTTPError(f'{env_prefix}_EXTRA_HEADERS keys must be strings')
376        normalized[key] = str(value)
377    return normalized

380def _supports_response_format_fallback(exc: LLMHTTPError) -> bool:
381    if exc.status_code < 400 or exc.status_code >= 500:
382        return False
383    message = str(exc).lower()
384    if 'response_format' not in message:
385        return False
386    return (
387        'unavailable' in message
388        or 'unsupported' in message
389        or 'not supported' in message
390        or 'invalid_request_error' in message
391    )

511        output_tokens = 0
512
513        def _handle(parsed: dict[str, Any]) -> None:

515            event_type = parsed.get('type', '')

517            if isinstance(usage, dict):

519                output_tokens = usage.get('output_tokens', output_tokens)

521            if event_type == 'content_block_delta' or event_type == 'message_delta':
522                delta = parsed.get('delta', {})

524                    text = str(delta.get('text', ''))

526                request.on_chunk(text)
527            if text:
528                chunks.append(text)
529
530        consume_sse_json_chunks(self._iter_streaming_lines(payload), on_data=_handle)
531        return LLMResponse(
532            provider=self.provider,
533            model=model,
534            content=''.join(chunks),
535            raw={},
536            input_tokens=input_tokens,
537            output_tokens=output_tokens,
538        )

541def _extract_gemini_stream_text(parsed: dict[str, Any]) -> str:
542    parts: list[str] = []
543    for candidate in parsed.get('candidates', []):
544        if not isinstance(candidate, dict):
545            continue
546        content = candidate.get('content', {})
547        if not isinstance(content, dict):
548            continue
549        for part in content.get('parts', []):
550            if isinstance(part, dict) and isinstance(part.get('text'), str):
551                parts.append(part['text'])
552    return ''.join(parts)

573    def complete(self, request: LLMRequest) -> LLMResponse:
574        model = request.model or self.config.resolved_model()
575        contents = []
576        for message in request.messages:
577            role = 'model' if message.role == 'assistant' else 'user'
578            contents.append({'role': role, 'parts': [{'text': message.content}]})
579        payload: dict[str, Any] = {
580            'contents': contents,
581            'generationConfig': {
582                'maxOutputTokens': request.max_tokens,
583                'temperature': request.temperature,
584            },
585        }
586        if request.system:
587            payload['systemInstruction'] = {'parts': [{'text': request.system}]}
588        if request.tools:
589            payload['tools'] = [
590                {
591                    'functionDeclarations': [
592                        {
593                            'name': t.name,
594                            'description': t.description,
595                            'parameters': t.input_schema,
596                        }
597                    ]
598                }
599                for t in request.tools
600            ]
601        if request.stream:
602            return self._complete_streaming(request, model, payload)
603        response = _call_with_retry(
604            self.provider,
605            lambda: self.transport.post_json(
606                f'{self.config.resolved_base_url()}/models/{model}:generateContent?key={self.config.resolved_api_key()}',
607                {},
608                payload,
609                timeout=self.timeout,
610            ),
611            self.retry_config,
612        )
613        content = _extract_gemini_content(response)
614        metadata = response.get('usageMetadata', {})
615        tool_calls = _extract_gemini_tool_calls(response)
616        safety = _extract_gemini_safety(response)
617        return LLMResponse(
618            provider=self.provider,
619            model=model,
620            content=content,
621            raw=response,
622            input_tokens=metadata.get('promptTokenCount', 0),
623            output_tokens=metadata.get('candidatesTokenCount', 0),
624            tool_calls=tool_calls,
625            safety=safety,
626        )

646    def _complete_streaming(
647        self, request: LLMRequest, model: str, payload: dict[str, Any]
648    ) -> LLMResponse:
649        chunks: list[str] = []
650        input_tokens = 0
651        output_tokens = 0
652
653        def _handle(parsed: dict[str, Any]) -> None:
654            nonlocal input_tokens, output_tokens
655            metadata = parsed.get('usageMetadata', {})
656            if isinstance(metadata, dict):
