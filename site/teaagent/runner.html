 31    """Executes an agent run loop: decide, dispatch tools, enforce budgets, record audit events.
 32
 33    The runner orchestrates the core agent lifecycle:
 34    1. Calls the *decide* function with the current context.
 35    2. On a ``FinalAnswer``, records ``run_completed`` and returns.
 36    3. On a tool request, validates tool existence, checks policy, dispatches the tool,
 37       records the observation, and loops.
 38    4. On budget exhaustion, records ``run_failed`` with a budget‑exceeded error.
 39    """

177                annotations = {
178                    'read_only': tool.annotations.read_only,
179                    'destructive': tool.annotations.destructive,

197                if self.auto_mode_guard is not None:

234                        secret = (

236                        )
237                    approval_request = ApprovalRequest(
238                        call_id=decision.call_id,
239                        tool_name=decision.tool_name,
240                        arguments=decision.arguments,
241                        reason=str(exc),
242                        annotations=annotations,
243                        run_id=current_run_id,
244                        workspace_secret=secret,
245                    )

247                        pending_payload = approval_request.to_dict()
248                        pending_payload.pop('run_id', None)

253                        )
254                        if self.approval_handler is None:
255                            self.audit.record(
256                                'run_paused',
257                                current_run_id,
258                                status='pending_approval',
259                                approval=approval_request.to_dict(),
260                                cost_cents=cost_cents,
261                            )
262                            if self.checkpoint_store is not None:

266                                final_answer=None,
267                                iterations=iterations,
268                                tool_calls=tool_calls,
269                                status='pending_approval',
270                                metadata={'approval': approval_request.to_dict()},
271                                cost_cents=cost_cents,
272                                input_tokens=input_tokens,
273                                output_tokens=output_tokens,

275                        if self.approval_handler(approval_request):

300                    'tool_call_started',
301                    current_run_id,
302                    call_id=decision.call_id,
303                    tool_name=decision.tool_name,
304                    arguments=decision.arguments,
305                    annotations=annotations,
306                )

312                        )
313                    finally:

318                        'call_id': decision.call_id,
319                        'tool_name': decision.tool_name,
320                        'error': str(exc),
321                    }
322                    context['observations'].append(err_observation)
323                    self.audit.record(
324                        'tool_call_failed', current_run_id, **err_observation
325                    )
326                    if self.checkpoint_store is not None:

328                    continue

331                    self.auto_mode_guard.record_tool_call()
332                observation = {
333                    'call_id': decision.call_id,
334                    'tool_name': decision.tool_name,
335                    'result': result,
336                }
337                context['observations'].append(observation)
338                self.audit.record('tool_call_completed', current_run_id, **observation)
339                if self.checkpoint_store is not None:
340                    self.checkpoint_store.save(current_run_id, context)

342                    self.compactor

359                    input_tokens=input_tokens,
360                    output_tokens=output_tokens,
361                )
362                return RunResult(
363                    run_id=current_run_id,
364                    final_answer=None,
365                    iterations=iterations,
366                    tool_calls=tool_calls,
367                    status=f'failed:{exc.category}',

369                    cost_cents=cost_cents,

399            message='iteration budget exceeded',
400            cost_cents=cost_cents,
401            input_tokens=input_tokens,
402            output_tokens=output_tokens,
403        )

Executes an agent run loop: decide, dispatch tools, enforce budgets, record audit events.

The runner orchestrates the core agent lifecycle:

  1. Calls the decide function with the current context.
  2. On a FinalAnswer, records run_completed and returns.
  3. On a tool request, validates tool existence, checks policy, dispatches the tool, records the observation, and loops.
  4. On budget exhaustion, records run_failed with a budget‑exceeded error.

216                        raise ToolPermissionError(
217                            'Tool registry has lint errors; read-only runs cannot '
218                            f'invoke tools ({len(self._read_only_registry_lint_errors)} error(s))'
219                        )
220                    self.approval_policy.assert_allowed(
221                        tool_name=decision.tool_name,
222                        call_id=decision.call_id,
223                        destructive=tool.annotations.destructive,
224                        arguments=decision.arguments,
225                        jit_state=self.jit_state,
226                        plan_contract=plan_contract,
227                        read_only=tool.annotations.read_only,
228                        description=tool.description,
229                        handler=tool.handler,
230                    )
231                except ToolPermissionError as exc:
232                    secret = None
233                    if self.approval_policy and self.approval_policy.approval_store:
234                        secret = (
235                            self.approval_policy.approval_store._get_workspace_secret()
236                        )
237                    approval_request = ApprovalRequest(
238                        call_id=decision.call_id,
239                        tool_name=decision.tool_name,
240                        arguments=decision.arguments,
241                        reason=str(exc),
242                        annotations=annotations,
243                        run_id=current_run_id,
244                        workspace_secret=secret,
245                    )
246                    if self._can_request_approval(tool.annotations.destructive):
247                        pending_payload = approval_request.to_dict()
248                        pending_payload.pop('run_id', None)
249                        self.audit.record(
250                            'tool_call_pending_approval',
251                            current_run_id,
252                            **pending_payload,
253                        )
254                        if self.approval_handler is None:
255                            self.audit.record(
256                                'run_paused',
257                                current_run_id,
258                                status='pending_approval',
259                                approval=approval_request.to_dict(),
260                                cost_cents=cost_cents,
261                            )
262                            if self.checkpoint_store is not None:
263                                self.checkpoint_store.save(current_run_id, context)
264                            return RunResult(
265                                run_id=current_run_id,
266                                final_answer=None,
267                                iterations=iterations,
268                                tool_calls=tool_calls,
269                                status='pending_approval',
270                                metadata={'approval': approval_request.to_dict()},
271                                cost_cents=cost_cents,
272                                input_tokens=input_tokens,
273                                output_tokens=output_tokens,
274                            )
275                        if self.approval_handler(approval_request):
276                            self.audit.record(
277                                'tool_call_approved',
278                                current_run_id,
279                                call_id=decision.call_id,
280                                tool_name=decision.tool_name,
281                            )
282                        else:
283                            self.audit.record(
284                                'tool_call_denied',
285                                current_run_id,
286                                call_id=decision.call_id,
287                                tool_name=decision.tool_name,
288                            )
289                            raise
290                    else:
291                        blocked_payload = approval_request.to_dict()
292                        blocked_payload.pop('run_id', None)
293                        self.audit.record(
294                            'tool_call_blocked',
295                            current_run_id,
296                            **blocked_payload,
297                        )
298                        raise
299                self.audit.record(
300                    'tool_call_started',
301                    current_run_id,
302                    call_id=decision.call_id,
303                    tool_name=decision.tool_name,
304                    arguments=decision.arguments,
305                    annotations=annotations,
306                )
307                try:
308                    parent_token = bind_parent_run_id(current_run_id)
309                    try:
310                        result = self.registry.execute(
311                            decision.tool_name, decision.arguments
312                        )
313                    finally:
314                        reset_parent_run_id(parent_token)
315                except ToolExecutionError as exc:
316                    tool_calls += 1
317                    err_observation: dict[str, Any] = {
318                        'call_id': decision.call_id,
319                        'tool_name': decision.tool_name,
320                        'error': str(exc),
321                    }
322                    context['observations'].append(err_observation)
323                    self.audit.record(
324                        'tool_call_failed', current_run_id, **err_observation
325                    )
326                    if self.checkpoint_store is not None:
327                        self.checkpoint_store.save(current_run_id, context)
328                    continue
329                tool_calls += 1
330                if self.auto_mode_guard is not None:
331                    self.auto_mode_guard.record_tool_call()
332                observation = {
333                    'call_id': decision.call_id,
334                    'tool_name': decision.tool_name,
335                    'result': result,
336                }
337                context['observations'].append(observation)
338                self.audit.record('tool_call_completed', current_run_id, **observation)
339                if self.checkpoint_store is not None:
340                    self.checkpoint_store.save(current_run_id, context)
341                if (
342                    self.compactor
343                    and len(context['observations']) > self.compact_after_observations
344                ):
345                    compacted = self.compactor.compact(context)
346                    context['observations'] = compacted.context['observations']
347                    context['compacted_summary'] = compacted.summary
348                    context['memory_keys'] = compacted.pinned
349                    self.audit.record(
350                        'context_compacted', current_run_id, summary=compacted.summary
351                    )
352            except AgentHarnessError as exc:
353                self.audit.record(
354                    'run_failed',
355                    current_run_id,
356                    category=exc.category,
357                    message=str(exc),
358                    cost_cents=cost_cents,
359                    input_tokens=input_tokens,
360                    output_tokens=output_tokens,
361                )
362                return RunResult(
363                    run_id=current_run_id,
364                    final_answer=None,
365                    iterations=iterations,
366                    tool_calls=tool_calls,
367                    status=f'failed:{exc.category}',
368                    error_message=str(exc),
369                    cost_cents=cost_cents,
370                    input_tokens=input_tokens,
371                    output_tokens=output_tokens,
372                )
373            except Exception as exc:  # pragma: no cover - defensive boundary
374                self.audit.record(
375                    'run_failed',
376                    current_run_id,
377                    category=ErrorCategory.SYSTEM,
378                    message=str(exc),
379                    cost_cents=cost_cents,
380                    input_tokens=input_tokens,
381                    output_tokens=output_tokens,
382                )
383                return RunResult(
384                    run_id=current_run_id,
385                    final_answer=None,
386                    iterations=iterations,
387                    tool_calls=tool_calls,
388                    status=f'failed:{ErrorCategory.SYSTEM}',
389                    error_message=str(exc),
390                    cost_cents=cost_cents,
391                    input_tokens=input_tokens,
392                    output_tokens=output_tokens,
393                )
394
395        self.audit.record(
396            'run_failed',
397            current_run_id,
398            category=ErrorCategory.MODEL_LOGIC,
399            message='iteration budget exceeded',
400            cost_cents=cost_cents,
401            input_tokens=input_tokens,
402            output_tokens=output_tokens,
403        )
404        return RunResult(
405            run_id=current_run_id,
406            final_answer=None,
407            iterations=iterations,
408            tool_calls=tool_calls,
409            status=f'failed:{ErrorCategory.MODEL_LOGIC}',
410            error_message='iteration budget exceeded',
411            cost_cents=cost_cents,
412            input_tokens=input_tokens,
413            output_tokens=output_tokens,
414        )
