 40    def handle_event(self, event: Any) -> None:
 41        etype: str = event.event_type
 42        run_id: str = event.run_id
 43        payload: dict[str, Any] = getattr(event, 'payload', {})
 44
 45        if etype == 'run_started':
 46            span = self._tracer.start_span('agent.run')
 47            span.set_attribute('agent.task', payload.get('task', ''))
 48            span.set_attribute('agent.run_id', run_id)
 49            self._run_spans[run_id] = span
 50
 51        elif etype == 'tool_call_started':
 52            call_id = payload.get('call_id', '')
 53            name = payload.get('tool_name', 'unknown')
 54            annotations = payload.get('annotations', {})
 55
 56            parent_span = self._run_spans.get(run_id)
 57            ctx = (
 58                _otel_trace.set_span_in_context(parent_span)
 59                if parent_span is not None
 60                else None
 61            )
 62            span = self._tracer.start_span(
 63                'tool.call',
 64                context=ctx,
 65                attributes={
 66                    'tool.name': name,
 67                    'tool.destructive': annotations.get('destructive', False),
 68                    'tool.idempotent': annotations.get('idempotent', False),
 69                    'tool.read_only': annotations.get('read_only', True),
 70                },
 71            )
 72            self._tool_spans[call_id] = span
 73
 74        elif etype == 'tool_call_completed':
 75            call_id = payload.get('call_id', '')
 76            tool_span: Optional['_otel_trace.Span'] = self._tool_spans.get(call_id)
 77            if tool_span is not None:
 78                tool_span.end()
 79                del self._tool_spans[call_id]
 80
 81        elif etype in ('run_completed', 'run_failed'):
 82            run_span: Optional['_otel_trace.Span'] = self._run_spans.get(run_id)
 83            if run_span is not None:
 84                del self._run_spans[run_id]
 85                if etype == 'run_failed':
 86                    run_span.set_status(
 87                        Status(StatusCode.ERROR, payload.get('error', ''))
 88                    )
 89                else:
 90                    run_span.set_status(Status(StatusCode.OK))
 91                run_span.set_attribute('agent.outcome', etype)
 92                run_span.set_attribute('agent.iterations', payload.get('iterations', 0))
 93                run_span.end()
 94
 95    def shutdown(self) -> None:
 96        for span in self._run_spans.values():
 97            if span.is_recording():
 98                span.set_status(Status(StatusCode.ERROR, 'agent interrupted'))
 99                span.end()
100        for span in self._tool_spans.values():
101            if span.is_recording():
102                span.end()
103        self._run_spans.clear()
104        self._tool_spans.clear()

 99        if etype in {'run_started', 'run_completed', 'run_failed'}:
100            self._run_counter.add(1, {'event_type': etype})
101            self._record_number(self._iteration_histogram, payload, 'iterations')
102            self._record_number(self._cost_histogram, payload, 'cost_cents')
103        elif etype in {'tool_call_started', 'tool_call_completed'}:
104            self._tool_call_counter.add(
