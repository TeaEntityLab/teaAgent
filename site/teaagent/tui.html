 278    def run(self, *, run_setup: bool = False, setup_write_env: bool = False) -> int:
 279        self._load_workspace_defaults()
 280        self._load_tui_state()
 281        self._print_header()
 282        if run_setup:
 283            from teaagent.tui._setup import run_tui_setup
 284
 285            run_tui_setup(self, write_env=setup_write_env)
 286
 287        # Check if we should use split-pane layout
 288        use_split_pane = self._should_use_split_pane()
 289        if use_split_pane:
 290            self.output_fn(
 291                '[TeaAgent] Split-pane layout enabled (terminal size >= 120x30)'
 292            )
 293
 294        # Initialize prompt_toolkit session if available and no custom input_fn is provided
 295        if self.input_fn is None:
 296            try:
 297                from prompt_toolkit import PromptSession
 298                from prompt_toolkit.auto_suggest import AutoSuggestFromHistory
 299                from prompt_toolkit.history import FileHistory
 300
 301                from teaagent.tui._completion import TeaAgentCompleter
 302
 303                history_path = self._state_path.parent / 'history.txt'
 304                history_path.parent.mkdir(parents=True, exist_ok=True)
 305                self._session = PromptSession(
 306                    history=FileHistory(str(history_path)),
 307                    auto_suggest=AutoSuggestFromHistory(),
 308                    completer=TeaAgentCompleter(root=self.root),
 309                )
 310            except (ImportError, OSError):
 311                self._session = None
 312
 313        # Auto-start file watcher for pinned files
 314        self._start_file_watcher()
 315
 316        while True:
 317            try:
 318                if use_split_pane:
 319                    self._print_state_panel()
 320
 321                if self.input_fn:
 322                    raw_command = self.input_fn(self._prompt())
 323                elif self._session:
 324                    raw_command = self._session.prompt(self._prompt())
 325                else:
 326                    raw_command = input(self._prompt())
 327            except (EOFError, KeyboardInterrupt):
 328                self._stop_file_watcher()
 329                self.output_fn('bye')
 330                self._save_tui_state()
 331                return 0
 332
 333            should_continue = self.handle_command(raw_command)
 334            if not should_continue:
 335                self._stop_file_watcher()
 336                self._save_tui_state()
 337                return 0

 356    def _handle_memory(self, args: list[str]) -> None:
 357        if not args:
 358            self.output_fn('error: memory requires add, list, search, or show')
 359            return
 360        catalog = MemoryCatalog(self.root)
 361        action = args[0]
 362        rest = args[1:]
 363        if action == 'add':
 364            if not rest:
 365                self.output_fn('error: memory add requires text')
 366                return
 367            self._print_json(catalog.add(' '.join(rest)).to_dict())
 368            return
 369        if action == 'list':
 370            self._print_json([entry.to_dict() for entry in catalog.list()])
 371            return
 372        if action == 'search':
 373            if not rest:
 374                self.output_fn('error: memory search requires a query')
 375                return
 376            self._print_json(
 377                [entry.to_dict() for entry in catalog.search(' '.join(rest))]
 378            )
 379            return
 380        if action == 'show':
 381            if len(rest) != 1:
 382                self.output_fn('error: memory show requires one id')
 383                return
 384            self._print_json(catalog.show(rest[0]).to_dict())
 385            return
 386        if action == 'failures':
 387            self._handle_memory_failures(rest)
 388            return
 389        if action == 'clear':
 390            self._handle_memory_clear(rest)
 391            return
 392        self.output_fn(f"error: unknown memory command '{action}'")

 394    def _handle_memory_failures(self, args: list[str]) -> None:
 395        try:
 396            from datetime import datetime
 397
 398            from teaagent.memory.failure_card import FailureCardStorage
 399
 400            storage = FailureCardStorage(self.root)
 401            cards = storage.list_all()
 402            if not cards:
 403                self.output_fn('error: no failure cards recorded')
 404                return
 405            self.output_fn(f'memory failures: {len(cards)} card(s)')
 406            for i, card in enumerate(cards, 1):
 407                ts = datetime.fromtimestamp(card.timestamp).strftime(
 408                    '%Y-%m-%d %H:%M:%S'
 409                )
 410                loc = (
 411                    f'{card.file_path}:{card.line_number}'
 412                    if card.line_number
 413                    else card.file_path or '?'
 414                )
 415                self.output_fn(
 416                    f'  [{i}] run #{card.run_id} {card.error_type} at {loc} ({ts})'
 417                )
 418                self.output_fn(f'       task: {card.task_description}')
 419                self.output_fn(f'       error: {card.error_message}')
 420        except Exception as exc:
 421            self.output_fn(f'error: memory failures: {exc}')

 508    def _on_file_changed(self, file_path: str, event_type: str) -> None:
 509        """Callback when a pinned file is modified or deleted."""
 510        try:
 511            from teaagent.memory.pinned_file import PinnedFileStorage
 512
 513            storage = PinnedFileStorage(self.root)
 514            if event_type == 'deleted':
 515                storage.remove(file_path)
 516                self.output_fn(f'file unpinned (deleted): {file_path}')
 517                pinned_files = storage.list_all()
 518                if not pinned_files:
 519                    self._stop_file_watcher()
 520                elif self._file_watcher:
 521                    self._file_watcher.update_watched_files(
 522                        {pf.file_path for pf in pinned_files}
 523                    )
 524            elif event_type == 'modified':
 525                storage.update_last_modified(file_path)
 526                self.output_fn(f'context refreshed: {file_path}')
 527        except Exception as exc:
 528            self.output_fn(f'warning: file change handler error: {exc}')
 529
 530    def _start_file_watcher(self) -> None:
 531        """Start the file watcher if there are pinned files."""
 532        if self._watcher_running:
 533            return
 534        try:
 535            from teaagent.memory.file_watcher import FileWatcher
 536            from teaagent.memory.pinned_file import PinnedFileStorage
 537
 538            storage = PinnedFileStorage(self.root)
 539            pinned = storage.list_all()
 540            if pinned:
 541                self._file_watcher = FileWatcher(
 542                    root=self.root,
 543                    callback=self._on_file_changed,
 544                    debounce_ms=500,
 545                )
 546                self._file_watcher.update_watched_files({pf.file_path for pf in pinned})
 547                self._file_watcher.start()
 548                self._watcher_running = True
 549                self.output_fn(f'watching {len(pinned)} pinned file(s)')
 550        except Exception as exc:
 551            self.output_fn(f'warning: file watcher start failed: {exc}')
 552
 553    def _stop_file_watcher(self) -> None:
 554        """Stop the file watcher."""
 555        if self._file_watcher and self._watcher_running:
 556            from contextlib import suppress
 557
 558            with suppress(Exception):
 559                self._file_watcher.stop()
 560            self._watcher_running = False
 561            self._file_watcher = None

 565    def _create_checkpoint(self) -> bool:
 566        """Create a git stash checkpoint to protect pre-session changes."""
 567        import subprocess
 568
 569        timestamp = __import__('time').time()
 570        self._checkpoint_ref = f'teaagent-checkpoint-{int(timestamp)}'
 571        try:
 572            result = subprocess.run(
 573                ['git', 'stash', 'push', '-m', self._checkpoint_ref],
 574                cwd=self.root,
 575                capture_output=True,
 576                text=True,
 577            )
 578            if result.returncode == 0:
 579                self._checkpoint_created = True
 580                self.output_fn(f'checkpoint created: {self._checkpoint_ref}')
 581                return True
 582            if 'No local changes to save' in result.stdout:
 583                self._checkpoint_created = True
 584                self.output_fn('checkpoint: clean workspace (no changes to stash)')
 585                return True
 586            self.output_fn(f'warning: checkpoint failed: {result.stderr}')
 587            return False

 595    def _restore_checkpoint(self) -> bool:
 596        """Restore checkpoint without destroying unstaged user edits."""
 597        import subprocess
 598
 599        if not self._checkpoint_created:
 600            self.output_fn('no checkpoint to restore')
 601            return False
 602        try:
 603            result = subprocess.run(
 604                ['git', 'stash', 'list'],
 605                cwd=self.root,
 606                capture_output=True,
 607                text=True,
 608            )
 609            stash_exists = (
 610                self._checkpoint_ref and self._checkpoint_ref in result.stdout
 611            )
 612            if not stash_exists:
 613                self.output_fn('no checkpoint stash found')
 614                return False
 615
 616            show_result = subprocess.run(
 617                ['git', 'stash', 'show', '--name-only', '--oneline'],
 618                cwd=self.root,
 619                capture_output=True,
 620                text=True,
 621            )
 622            if show_result.returncode != 0:
 623                self.output_fn('error: failed to list checkpoint files')
 624                return False
 625
 626            lines = show_result.stdout.strip().split('\n')
 627            stashed_files = [ln.strip() for ln in lines[1:] if ln.strip()]
 628
 629            if stashed_files:
 630                checkout_result = subprocess.run(
 631                    ['git', 'checkout', 'HEAD', '--', *stashed_files],
 632                    cwd=self.root,
 633                    capture_output=True,
 634                    text=True,
 635                )
 636                if checkout_result.returncode != 0:
 637                    self.output_fn(
 638                        f'error: failed to restore files: {checkout_result.stderr}'
 639                    )
 640                    return False
 641
 642            pop_result = subprocess.run(
 643                ['git', 'stash', 'pop'],
 644                cwd=self.root,
 645                capture_output=True,
 646                text=True,
 647            )
 648            if pop_result.returncode == 0:
 649                self.output_fn(f'checkpoint restored: {self._checkpoint_ref}')
 650            else:
 651                self.output_fn(
 652                    f'warning: stash pop had conflicts — '
 653                    f'checkpoint preserved as "{self._checkpoint_ref}"'
 654                )
 655                self.output_fn('  resolve manually or run: git stash drop')
 656            return True
 657        except FileNotFoundError:
 658            self.output_fn('error: git not found in PATH')
 659            return False
 660        except Exception as exc:
 661            self.output_fn(f'error: checkpoint restore failed: {exc}')
 662            return False

 880                    request.arguments.get('path')

 884                )

 886                store.grant(
 887                    request.tool_name,
 888                    scope='session',
 889                    permission_mode=self.permission_mode.value,
 890                    path_globs=[str(path)],
 891                    ttl_hours=8.0,
 892                )
 893                self.output_fn(
 894                    f'approval: registered session grant for {request.tool_name} matching path: {path}'
 895                )
 896            else:
 897                store.grant(
 898                    request.tool_name,
 899                    scope='session',
 900                    permission_mode=self.permission_mode.value,
 901                    ttl_hours=8.0,
 902                )
 903                self.output_fn(
 904                    f'approval: registered global session grant for {request.tool_name}'
 905                )

 907        elif answer == 't':

 909                request.tool_name,
 910                scope='session',

 279    def run(self, *, run_setup: bool = False, setup_write_env: bool = False) -> int:
 280        self._load_workspace_defaults()
 281        self._load_tui_state()
 282        self._print_header()
 283        if run_setup:

 285
 286            run_tui_setup(self, write_env=setup_write_env)

 288        # Check if we should use split-pane layout
 289        use_split_pane = self._should_use_split_pane()
 290        if use_split_pane:
 291            self.output_fn(

 293            )
 294
 295        # Initialize prompt_toolkit session if available and no custom input_fn is provided
 296        if self.input_fn is None:
 297            try:
 298                from prompt_toolkit import PromptSession
 299                from prompt_toolkit.auto_suggest import AutoSuggestFromHistory
 300                from prompt_toolkit.history import FileHistory
 301
 302                from teaagent.tui._completion import TeaAgentCompleter
 303
 304                history_path = self._state_path.parent / 'history.txt'
 305                history_path.parent.mkdir(parents=True, exist_ok=True)
 306                self._session = PromptSession(
 307                    history=FileHistory(str(history_path)),
 308                    auto_suggest=AutoSuggestFromHistory(),
 309                    completer=TeaAgentCompleter(root=self.root),
 310                )
 311            except (ImportError, OSError):
 312                self._session = None
 313
 314        # Auto-start file watcher for pinned files

 316
 317        while True:

 332                return 0
 333
 334            should_continue = self.handle_command(raw_command)
 335            if not should_continue:
 336                self._stop_file_watcher()
 337                self._save_tui_state()
 338                return 0

 357    def _handle_memory(self, args: list[str]) -> None:
 358        if not args:
 359            self.output_fn('error: memory requires add, list, search, or show')
 360            return
 361        catalog = MemoryCatalog(self.root)
 362        action = args[0]
 363        rest = args[1:]
 364        if action == 'add':
 365            if not rest:
 366                self.output_fn('error: memory add requires text')
 367                return
 368            self._print_json(catalog.add(' '.join(rest)).to_dict())
 369            return
 370        if action == 'list':
 371            self._print_json([entry.to_dict() for entry in catalog.list()])
 372            return
 373        if action == 'search':
 374            if not rest:
 375                self.output_fn('error: memory search requires a query')
 376                return
 377            self._print_json(
 378                [entry.to_dict() for entry in catalog.search(' '.join(rest))]
 379            )
 380            return
 381        if action == 'show':
 382            if len(rest) != 1:
 383                self.output_fn('error: memory show requires one id')
 384                return
 385            self._print_json(catalog.show(rest[0]).to_dict())
 386            return
 387        if action == 'failures':
 388            self._handle_memory_failures(rest)
 389            return
 390        if action == 'clear':
 391            self._handle_memory_clear(rest)
 392            return
 393        self.output_fn(f"error: unknown memory command '{action}'")

 509    def _on_file_changed(self, file_path: str, event_type: str) -> None:
 510        """Callback when a pinned file is modified or deleted."""
 511        try:
 512            from teaagent.memory.pinned_file import PinnedFileStorage
 513
 514            storage = PinnedFileStorage(self.root)
 515            if event_type == 'deleted':
 516                storage.remove(file_path)
 517                self.output_fn(f'file unpinned (deleted): {file_path}')
 518                pinned_files = storage.list_all()
 519                if not pinned_files:
 520                    self._stop_file_watcher()
 521                elif self._file_watcher:
 522                    self._file_watcher.update_watched_files(
 523                        {pf.file_path for pf in pinned_files}
 524                    )
 525            elif event_type == 'modified':
 526                storage.update_last_modified(file_path)
 527                self.output_fn(f'context refreshed: {file_path}')
 528        except Exception as exc:
 529            self.output_fn(f'warning: file change handler error: {exc}')

 531    def _start_file_watcher(self) -> None:

 566    def _create_checkpoint(self) -> bool:
 567        """Create a git stash checkpoint to protect pre-session changes."""
 568        import subprocess
 569
 570        timestamp = __import__('time').time()
 571        self._checkpoint_ref = f'teaagent-checkpoint-{int(timestamp)}'
 572        try:
 573            result = subprocess.run(
 574                ['git', 'stash', 'push', '-m', self._checkpoint_ref],
 575                cwd=self.root,
 576                capture_output=True,
 577                text=True,
 578            )
 579            if result.returncode == 0:
 580                self._checkpoint_created = True
 581                self.output_fn(f'checkpoint created: {self._checkpoint_ref}')
 582                return True
 583            if 'No local changes to save' in result.stdout:
 584                self._checkpoint_created = True
 585                self.output_fn('checkpoint: clean workspace (no changes to stash)')
 586                return True
 587            self.output_fn(f'warning: checkpoint failed: {result.stderr}')
 588            return False
 589        except FileNotFoundError:
 590            self.output_fn('error: git not found in PATH')
 591            return False
 592        except Exception as exc:
 593            self.output_fn(f'error: checkpoint failed: {exc}')
 594            return False
 595
 596    def _restore_checkpoint(self) -> bool:
 597        """Restore checkpoint without destroying unstaged user edits."""
 598        import subprocess
 599
 600        if not self._checkpoint_created:
 601            self.output_fn('no checkpoint to restore')
 602            return False
 603        try:
 604            result = subprocess.run(
 605                ['git', 'stash', 'list'],
 606                cwd=self.root,
 607                capture_output=True,
 608                text=True,
 609            )
 610            stash_exists = (
 611                self._checkpoint_ref and self._checkpoint_ref in result.stdout
 612            )
 613            if not stash_exists:
 614                self.output_fn('no checkpoint stash found')
 615                return False
 616
 617            show_result = subprocess.run(
 618                ['git', 'stash', 'show', '--name-only', '--oneline'],
 619                cwd=self.root,
 620                capture_output=True,
 621                text=True,
 622            )
 623            if show_result.returncode != 0:
 624                self.output_fn('error: failed to list checkpoint files')
 625                return False
 626
 627            lines = show_result.stdout.strip().split('\n')
 628            stashed_files = [ln.strip() for ln in lines[1:] if ln.strip()]
 629
 630            if stashed_files:
 631                checkout_result = subprocess.run(
 632                    ['git', 'checkout', 'HEAD', '--', *stashed_files],
 633                    cwd=self.root,
 634                    capture_output=True,
 635                    text=True,
 636                )
 637                if checkout_result.returncode != 0:
 638                    self.output_fn(
 639                        f'error: failed to restore files: {checkout_result.stderr}'
 640                    )
 641                    return False
 642
 643            pop_result = subprocess.run(
 644                ['git', 'stash', 'pop'],
 645                cwd=self.root,
 646                capture_output=True,
 647                text=True,
 648            )
 649            if pop_result.returncode == 0:
 650                self.output_fn(f'checkpoint restored: {self._checkpoint_ref}')
 651            else:
 652                self.output_fn(
 653                    f'warning: stash pop had conflicts — '
 654                    f'checkpoint preserved as "{self._checkpoint_ref}"'
 655                )
 656                self.output_fn('  resolve manually or run: git stash drop')
 657            return True
 658        except FileNotFoundError:
 659            self.output_fn('error: git not found in PATH')
 660            return False
 661        except Exception as exc:
 662            self.output_fn(f'error: checkpoint restore failed: {exc}')
 663            return False
