 30@dataclass
 31class _EndpointCircuit:
 32    failures: int = 0
 33    opened_at: Optional[float] = None  # monotonic timestamp when circuit opened
 34
 35    def is_open(self, reset_timeout: float) -> bool:
 36        if self.opened_at is None:
 37            return False
 38        return (_time.monotonic() - self.opened_at) < reset_timeout
 39
 40    def record_failure(self, threshold: int) -> None:
 41        self.failures += 1
 42        if self.failures >= threshold:
 43            self.opened_at = _time.monotonic()
 44
 45    def record_success(self) -> None:
 46        self.failures = 0
 47        self.opened_at = None

136    def __init__(self, path: str | Path) -> None:
137        self._path = Path(path)
138        self._path.parent.mkdir(parents=True, exist_ok=True)
139        self._init_db()
140
141    def _connect(self) -> sqlite3.Connection:
142        conn = sqlite3.connect(str(self._path), timeout=10)
143        conn.execute('PRAGMA journal_mode=WAL')
144        return conn
145
146    def _init_db(self) -> None:
147        with self._connect() as conn:
148            conn.execute(
149                """
150                CREATE TABLE IF NOT EXISTS agent_cards (
151                    name TEXT PRIMARY KEY,
152                    card_json TEXT NOT NULL,
153                    registered_at TEXT NOT NULL
154                )
155                """
156            )
157
158    def register(self, card: AgentCard) -> None:
159        now = datetime.now(timezone.utc).isoformat()
160        with self._connect() as conn:
161            conn.execute(
162                """
163                INSERT INTO agent_cards (name, card_json, registered_at)
164                VALUES (?, ?, ?)
165                ON CONFLICT(name) DO UPDATE SET
166                    card_json = excluded.card_json,
167                    registered_at = excluded.registered_at
168                """,
169                (card.name, json.dumps(card.to_dict(), ensure_ascii=False), now),
170            )
171
172    def deregister(self, name: str) -> None:
173        with self._connect() as conn:
174            conn.execute('DELETE FROM agent_cards WHERE name = ?', (name,))
175
176    def get(self, name: str) -> Optional[AgentCard]:
177        with self._connect() as conn:
178            row = conn.execute(
179                'SELECT card_json FROM agent_cards WHERE name = ?', (name,)
180            ).fetchone()
181        if row is None:
182            return None
183        return AgentCard.from_dict(json.loads(row[0]))
184
185    def list_cards(self) -> list[AgentCard]:
186        with self._connect() as conn:
187            rows = conn.execute('SELECT card_json FROM agent_cards').fetchall()
188        return [AgentCard.from_dict(json.loads(r[0])) for r in rows]
189
190    def find_by_capability(self, capability: str) -> list[AgentCard]:
191        return [c for c in self.list_cards() if capability in c.capabilities]

243def _make_a2a_handler(
244    card_data: dict[str, Any],
245    task_handler: Optional[Callable[[str, dict[str, Any]], str]],
246) -> type:
247    class _Handler(http.server.BaseHTTPRequestHandler):
248        def do_GET(self) -> None:
249            if self.path == '/.well-known/agent.json':
250                body = json.dumps(card_data).encode('utf-8')
251                self.send_response(200)
252                self.send_header('Connection', 'close')
253                self.send_header('Content-Type', 'application/json')
254                self.send_header('Content-Length', str(len(body)))
255                self.end_headers()
256                self.wfile.write(body)
257            else:
258                self.send_response(404)
259                self.send_header('Connection', 'close')
260                self.end_headers()
261
262        def do_POST(self) -> None:
263            if self.path == '/a2a/task' and task_handler is not None:
264                length = int(self.headers.get('Content-Length', '0'))
265                body = self.rfile.read(length)
266                try:
267                    req_data = json.loads(body.decode('utf-8'))
268                    output = task_handler(
269                        req_data.get('task', ''), req_data.get('context') or {}
270                    )
271                    resp_body = json.dumps(
272                        {'agent_name': card_data.get('name', ''), 'output': output}
273                    ).encode('utf-8')
274                    self.send_response(200)
275                    self.send_header('Connection', 'close')
276                    self.send_header('Content-Type', 'application/json')
277                    self.send_header('Content-Length', str(len(resp_body)))
278                    self.end_headers()
279                    self.wfile.write(resp_body)
280                except Exception as exc:
281                    resp_body = json.dumps({'error': str(exc)}).encode('utf-8')
282                    self.send_response(500)
283                    self.send_header('Connection', 'close')
284                    self.send_header('Content-Type', 'application/json')
285                    self.send_header('Content-Length', str(len(resp_body)))
286                    self.end_headers()
287                    self.wfile.write(resp_body)
288            else:
289                self.send_response(404)
290                self.send_header('Connection', 'close')
291                self.end_headers()
292
293        def log_message(self, *_: object) -> None:
294            pass
295
296    return _Handler

299class A2ADiscoveryServer:
300    """Serves an AgentCard at ``/.well-known/agent.json`` over HTTP.
301
302    Optionally handles POST ``/a2a/task`` for in-process task delegation when
303    *task_handler* is provided.  Use ``port=0`` to let the OS pick a free port
304    (useful in tests).
305    """
306
307    WELL_KNOWN_PATH = '/.well-known/agent.json'
308    TASK_PATH = '/a2a/task'
309
310    def __init__(
311        self,
312        card: AgentCard,
313        *,
314        host: str = '127.0.0.1',
315        port: int = 0,
316        task_handler: Optional[Callable[[str, dict[str, Any]], str]] = None,
317    ) -> None:
318        self._card = card
319        self._host = host
320        self._port = port
321        self._task_handler = task_handler
322        self._server: Optional[http.server.HTTPServer] = None
323        self._thread: Optional[threading.Thread] = None
324
325    @property
326    def port(self) -> int:
327        if self._server is not None:
328            return self._server.server_address[1]
329        return self._port
330
331    @property
332    def base_url(self) -> str:
333        return f'http://{self._host}:{self.port}'
334
335    def start(self) -> None:
336        card_data = self._card.to_dict()
337        handler_cls = _make_a2a_handler(card_data, self._task_handler)
338        self._server = http.server.HTTPServer((self._host, self._port), handler_cls)
339        if not card_data.get('endpoint'):
340            card_data['endpoint'] = self.base_url
341        self._thread = threading.Thread(target=self._server.serve_forever, daemon=True)
342        self._thread.start()
343
344    def stop(self) -> None:
345        if self._server is not None:
346            self._server.shutdown()
347            self._server = None
348
349    def __enter__(self) -> 'A2ADiscoveryServer':
350        self.start()
351        return self
352
353    def __exit__(self, *_: object) -> None:
354        self.stop()

357class A2AClient:
358    """HTTP client for A2A task delegation and card discovery."""
359
360    def __init__(self, endpoint: str, *, timeout: int = 30) -> None:
361        self._endpoint = endpoint.rstrip('/')
362        self._timeout = timeout
363
364    @classmethod
365    def from_card(cls, card: AgentCard, *, timeout: int = 30) -> 'A2AClient':
366        if not card.endpoint:
367            raise ValueError(f'AgentCard {card.name!r} has no endpoint')
368        return cls(card.endpoint, timeout=timeout)
369
370    def fetch_card(self) -> AgentCard:
371        url = f'{self._endpoint}/.well-known/agent.json'
372        with urllib.request.urlopen(url, timeout=self._timeout) as resp:
373            data = json.loads(resp.read().decode('utf-8'))
374        return AgentCard.from_dict(data)
375
376    def delegate(
377        self,
378        task: str,
379        *,
380        context: Optional[dict[str, Any]] = None,
381        traceparent: Optional[str] = None,
382    ) -> A2ATaskResult:
383        payload = json.dumps({'task': task, 'context': context or {}}).encode('utf-8')
384        headers: dict[str, str] = {'Content-Type': 'application/json'}
385        if traceparent is not None:
386            headers['traceparent'] = traceparent
387        req = urllib.request.Request(
388            f'{self._endpoint}/a2a/task',
389            data=payload,
390            headers=headers,
391            method='POST',
392        )
393        with urllib.request.urlopen(req, timeout=self._timeout) as resp:
394            result_data = json.loads(resp.read().decode('utf-8'))
395        agent_name = result_data.get('agent_name') or ''
396        return A2ATaskResult(
397            task=task,
398            agent_name=agent_name,
399            output=result_data.get('output', ''),
400            traceparent=traceparent,
401        )

404class FederatedAgentRegistry:
405    """Pulls AgentCards from remote ``/.well-known/agent.json`` endpoints.
406
407    Cards are cached for *ttl_seconds* (default 300).  The cache is
408    refreshed lazily on first access and whenever it becomes stale.
409    ``refresh()`` can also be called manually and returns a list of
410    per-endpoint error strings for diagnostic purposes.
411    """
412
413    def __init__(
414        self,
415        endpoints: list[str],
416        *,
417        ttl_seconds: int = 300,
418        timeout: int = 10,
419        circuit_breaker: Optional[CircuitBreakerConfig] = None,
420    ) -> None:
421        self._endpoints = list(endpoints)
422        self._ttl = ttl_seconds
423        self._timeout = timeout
424        self._cache: list[AgentCard] = []
425        self._fetched_at: float = 0.0
426        self._cb_config = circuit_breaker
427        self._circuits: dict[str, _EndpointCircuit] = {}
428
429    def _is_stale(self) -> bool:
430        return _time.monotonic() - self._fetched_at > self._ttl
431
432    def refresh(self) -> list[str]:
433        errors: list[str] = []
434        cards: list[AgentCard] = []
435        for base_url in self._endpoints:
436            circuit = self._circuits.setdefault(base_url, _EndpointCircuit())
437            if self._cb_config is not None and circuit.is_open(
438                self._cb_config.reset_timeout_seconds
439            ):
440                continue  # circuit open — skip this endpoint
441            url = base_url.rstrip('/') + '/.well-known/agent.json'
442            try:
443                with urllib.request.urlopen(url, timeout=self._timeout) as resp:
444                    data = json.loads(resp.read().decode('utf-8'))
445                cards.append(AgentCard.from_dict(data))
446                circuit.record_success()
447            except Exception as exc:
448                errors.append(f'{base_url}: {exc}')
449                if self._cb_config is not None:
450                    circuit.record_failure(self._cb_config.failure_threshold)
451        self._cache = cards
452        self._fetched_at = _time.monotonic()
453        return errors
454
455    def circuit_state(self, base_url: str) -> str:
456        """Return ``'open'`` or ``'closed'`` for the given endpoint."""
457        circuit = self._circuits.get(base_url)
458        if circuit is None:
459            return 'closed'
460        if self._cb_config is None:
461            return 'closed'
462        return (
463            'open'
464            if circuit.is_open(self._cb_config.reset_timeout_seconds)
465            else 'closed'
466        )
467
468    def _ensure_fresh(self) -> None:
469        if self._is_stale():
470            self.refresh()
471
472    def get(self, name: str) -> Optional[AgentCard]:
473        self._ensure_fresh()
474        for card in self._cache:
475            if card.name == name:
476                return card
477        return None
478
479    def list_cards(self) -> list[AgentCard]:
480        self._ensure_fresh()
481        return list(self._cache)
482
483    def find_by_capability(self, capability: str) -> list[AgentCard]:
484        return [c for c in self.list_cards() if capability in c.capabilities]
485
486    def find_by_tool(self, tool_name: str) -> list[AgentCard]:
487        return [c for c in self.list_cards() if tool_name in c.tools]

405class FederatedAgentRegistry:
406    """Pulls AgentCards from remote ``/.well-known/agent.json`` endpoints.
407
408    Cards are cached for *ttl_seconds* (default 300).  The cache is
409    refreshed lazily on first access and whenever it becomes stale.
410    ``refresh()`` can also be called manually and returns a list of
411    per-endpoint error strings for diagnostic purposes.
412    """
413
414    def __init__(
415        self,
416        endpoints: list[str],
417        *,
418        ttl_seconds: int = 300,
419        timeout: int = 10,
420        circuit_breaker: Optional[CircuitBreakerConfig] = None,
421    ) -> None:
422        self._endpoints = list(endpoints)
423        self._ttl = ttl_seconds
424        self._timeout = timeout
425        self._cache: list[AgentCard] = []
426        self._fetched_at: float = 0.0
427        self._cb_config = circuit_breaker
428        self._circuits: dict[str, _EndpointCircuit] = {}
429
430    def _is_stale(self) -> bool:
431        return _time.monotonic() - self._fetched_at > self._ttl
432
433    def refresh(self) -> list[str]:
434        errors: list[str] = []
435        cards: list[AgentCard] = []
436        for base_url in self._endpoints:
437            circuit = self._circuits.setdefault(base_url, _EndpointCircuit())
438            if self._cb_config is not None and circuit.is_open(
439                self._cb_config.reset_timeout_seconds
440            ):
441                continue  # circuit open — skip this endpoint
442            url = base_url.rstrip('/') + '/.well-known/agent.json'
443            try:
444                with urllib.request.urlopen(url, timeout=self._timeout) as resp:
445                    data = json.loads(resp.read().decode('utf-8'))
446                cards.append(AgentCard.from_dict(data))
447                circuit.record_success()
448            except Exception as exc:
449                errors.append(f'{base_url}: {exc}')
450                if self._cb_config is not None:
451                    circuit.record_failure(self._cb_config.failure_threshold)
452        self._cache = cards
453        self._fetched_at = _time.monotonic()
454        return errors
455
456    def circuit_state(self, base_url: str) -> str:
457        """Return ``'open'`` or ``'closed'`` for the given endpoint."""
458        circuit = self._circuits.get(base_url)
459        if circuit is None:
460            return 'closed'
461        if self._cb_config is None:
462            return 'closed'
463        return (
464            'open'
465            if circuit.is_open(self._cb_config.reset_timeout_seconds)
466            else 'closed'
467        )

469    def _ensure_fresh(self) -> None:
470        if self._is_stale():
471            self.refresh()
472
473    def get(self, name: str) -> Optional[AgentCard]:
474        self._ensure_fresh()

478        return None
